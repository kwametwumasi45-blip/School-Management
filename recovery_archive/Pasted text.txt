<script>
const descriptions = {
  "Teachers & Staff":"Manage teachers, staff details, and assignments.",
  "Classes & Subjects":"Organize classes, subjects, and class lists.",
  "Attendance":"Record and review student attendance.",
  "Fees & Payments":"Manage school fees, payments, and balances.",
  "Examinations":"Manage examination scores and student results.",
  "Reports":"Review school statistics and generate reports.",
  "Settings":"Configure school information and system preferences."
};

const $ = id => document.getElementById(id);
let editingId = null;
let students = [];
let searchTimer;

async function api(url, options = {}) {
  const response = await fetch(url, {
    ...options,
    headers: {
      ...(options.body ? {"Content-Type":"application/json"} : {}),
      ...options.headers
    }
  });

  const data = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(data.error || `Request failed (${response.status})`);
  return data;
}

function message(id, text, error = false) {
  $(id).textContent = text;
  $(id).classList.toggle("error", error);
}

function showPage(page) {
  const dashboard = page === "Dashboard";
  $("page-title").textContent = page;
  $("dashboard-page").hidden = !dashboard;
  $("module-page").hidden = dashboard;
  $("placeholder").hidden = page === "Students";
  $("students-page").hidden = page !== "Students";
  $("module-title").textContent = page;
  $("module-description").textContent = descriptions[page] || "";

  document.querySelectorAll("nav a").forEach(link => {
    link.classList.toggle("active", link.dataset.page === page);
  });

  if (page === "Students") loadStudents();
  if (page === "Dashboard") loadStudentCount();
}

function loadPageFromHash() {
  const hash = location.hash.slice(1).toLowerCase();
  const link = [...document.querySelectorAll("nav a")]
    .find(item => item.getAttribute("href") === "#" + hash);
  showPage(link ? link.dataset.page : "Dashboard");
}

async function loadStudentCount() {
  try {
    const data = await api("/api/students");
    $("student-count").textContent = data.length;
  } catch {
    $("student-count").textContent = "—";
  }
}

function addCell(row, value) {
  const cell = document.createElement("td");
  cell.textContent = value ?? "";
  row.appendChild(cell);
  return cell;
}

async function loadStudents() {
  $("student-table").replaceChildren();
  const loadingRow = $("student-table").insertRow();
  const loadingCell = loadingRow.insertCell();
  loadingCell.colSpan = 7;
  loadingCell.textContent = "Loading...";

  try {
    const query = $("student-search").value.trim();
    students = await api("/api/students?search=" + encodeURIComponent(query));
    renderStudents();
    await loadStudentCount();
  } catch (error) {
    $("student-table").replaceChildren();
    const row = $("student-table").insertRow();
    const cell = row.insertCell();
    cell.colSpan = 7;
    cell.textContent = "Could not load records.";
    message("list-message", error.message, true);
  }
}

function renderStudents() {
  const tbody = $("student-table");
  tbody.replaceChildren();

  if (!students.length) {
    const row = tbody.insertRow();
    const cell = row.insertCell();
    cell.colSpan = 7;
    cell.textContent = "No students found.";
    cell.style.textAlign = "center";
    $("record-count").textContent = "0 students";
    return;
  }

  for (const student of students) {
    const row = tbody.insertRow();
    addCell(row, student.student_id);
    addCell(row, `${student.first_name} ${student.last_name}`);
    addCell(row, student.gender);
    addCell(row, student.class_name);
    addCell(row, student.guardian_name);
    addCell(row, student.guardian_phone);

    const cell = row.insertCell();
    const actions = document.createElement("div");
    actions.className = "actions";

    const edit = document.createElement("button");
    edit.type = "button";
    edit.textContent = "Edit";
    edit.addEventListener("click", () => startEdit(student.id));

    const remove = document.createElement("button");
    remove.type = "button";
    remove.textContent = "Delete";
    remove.className = "danger";
    remove.addEventListener("click", () => deleteStudent(student.id));

    actions.append(edit, remove);
    cell.appendChild(actions);
  }

  $("record-count").textContent =
    `${students.length} student${students.length === 1 ? "" : "s"}`;
  message("list-message", "");
}

$("student-form").addEventListener("submit", async event => {
  event.preventDefault();
  const button = $("save-button");
  button.disabled = true;
  message("form-message", "");

  const body = Object.fromEntries(new FormData(event.currentTarget).entries());

  try {
    const result = await api(
      editingId === null ? "/api/students" : `/api/students/${editingId}`,
      {
        method: editingId === null ? "POST" : "PUT",
        body: JSON.stringify(body)
      }
    );

    message("form-message",
      editingId === null
        ? `Student ${result.student_id} registered successfully.`
        : `Student ${result.student_id} updated successfully.`
    );

    resetForm();
    await loadStudents();
  } catch (error) {
    message("form-message", error.message, true);
  } finally {
    button.disabled = false;
  }
});

function startEdit(id) {
  const student = students.find(item => item.id === id);
  if (!student) return;

  editingId = id;
  for (const field of [
    "student_id", "first_name", "last_name", "gender",
    "date_of_birth", "class_name", "guardian_name", "guardian_phone"
  ]) {
    $(field).value = student[field] || "";
  }

  $("form-heading").textContent = "Edit Student";
  $("save-button").textContent = "Save Changes";
  $("cancel-edit").hidden = false;
  message("form-message", "Editing " + student.student_id);
  $("student-form").scrollIntoView({behavior:"smooth", block:"start"});
}

function resetForm() {
  editingId = null;
  $("student-form").reset();
  $("form-heading").textContent = "Register New Student";
  $("save-button").textContent = "Register Student";
  $("cancel-edit").hidden = true;
}

$("cancel-edit").addEventListener("click", () => {
  resetForm();
  message("form-message", "");
});

async function deleteStudent(id) {
  const student = students.find(item => item.id === id);
  if (!student) return;

  if (!confirm(`Delete student ${student.student_id} (${student.first_name} ${student.last_name})? This cannot be undone.`)) return;

  try {
    await api(`/api/students/${id}`, {method:"DELETE"});
    if (editingId === id) resetForm();
    message("list-message", "Student deleted successfully.");
    await loadStudents();
  } catch (error) {
    message("list-message", error.message, true);
  }
}

$("search-button").addEventListener("click", loadStudents);
$("refresh-button").addEventListener("click", () => {
  $("student-search").value = "";
  loadStudents();
});
$("student-search").addEventListener("input", () => {
  clearTimeout(searchTimer);
  searchTimer = setTimeout(loadStudents, 300);
});
$("quick-students").addEventListener("click", () => {
  location.hash = "students";
});

window.addEventListener("hashchange", loadPageFromHash);
loadPageFromHash();
</script>
