  async function apiRequest(url, method = "GET", data = null) {
    const options = {
      method,
      credentials: "same-origin",
      headers: { "Accept": "application/json" }
    };

    if (data !== null) {
      options.headers["Content-Type"] = "application/json";
      options.body = JSON.stringify(data);
    }

    const response = await fetch(url, options);

    if (response.status === 401) {
      window.location.href = "/login";
      throw new Error("Your session has expired. Please log in again.");
    }

    const text = await response.text();
    let payload = {};

    try {
      payload = text ? JSON.parse(text) : {};
    } catch {
      throw new Error("The server returned an invalid response.");
    }

    if (!response.ok || payload.success === false) {
      throw new Error(
        payload.message || payload.error ||
        `Request failed (${response.status}).`
      );
    }

    return payload;
  }

  function extractRows(payload) {
    if (Array.isArray(payload)) return payload;
    if (Array.isArray(payload.data)) return payload.data;
    if (Array.isArray(payload.results)) return payload.results;
    if (Array.isArray(payload.records)) return payload.records;

    if (payload.data && typeof payload.data === "object") {
      return [payload.data];
    }

    return [];
  }

  async function saveStudent(event) {
    event.preventDefault();

    const form = $("student-form");
    const formData = new FormData(form);
    const student = {};

    [
      "student_id",
      "first_name",
      "last_name",
      "gender",
      "date_of_birth",
      "class_name",
      "guardian_name",
      "guardian_phone"
    ].forEach(field => {
      student[field] = String(formData.get(field) || "").trim();
    });

    const wasEditing = editingStudentId !== null;
    const button = $("student-save-button");

    button.disabled = true;
    button.textContent = "Saving...";

    try {
      if (wasEditing) {
        await apiRequest(
          `${API.students}/${editingStudentId}`,
          "PUT",
          student
        );
      } else {
        await apiRequest(API.students, "POST", student);
      }

      resetStudentForm();
      await loadModule("students");

      setMessage(
        wasEditing
          ? "Student record updated successfully."
          : "Student registered successfully.",
        "success"
      );

    } catch (error) {
      setMessage(error.message || "Could not save student.", "error");
    } finally {
      button.disabled = false;

      if (editingStudentId === null) {
        button.textContent = "Save Student";
      }
    }
  }

  function editStudent(id) {
    const student = lastRows.find(
      row => String(row.id) === String(id)
    );

    if (!student) {
      setMessage("Student record could not be found.", "error");
      return;
    }

    editingStudentId = student.id;

    const form = $("student-form");

    Object.keys(student).forEach(key => {
      const input = form.elements.namedItem(key);

      if (input) {
        input.value = student[key] ?? "";
      }
    });

    $("student-form-title").textContent = "Edit Student Record";
    $("student-save-button").textContent = "Update Student";
    $("student-cancel-button").hidden = false;

    form.scrollIntoView({
      behavior: "smooth",
      block: "start"
    });
  }

  async function deleteStudent(id) {
    const student = lastRows.find(
      row => String(row.id) === String(id)
    );

    if (!student) return;

    const confirmed = window.confirm(
      `Are you sure you want to delete ${student.first_name} ${student.last_name}?`
    );

    if (!confirmed) return;

    try {
      await apiRequest(`${API.students}/${id}`, "DELETE");

      await loadModule("students");

      setMessage(
        "Student record deleted successfully.",
        "success"
      );

    } catch (error) {
      setMessage(
        error.message || "Could not delete student.",
        "error"
      );
    }
  }

  // ==========================================
  // RECORDS TABLE
  // ==========================================

  function renderTable(rows) {
    const head = $("records-head");
    const body = $("records-body");

    head.replaceChildren();
    body.replaceChildren();

    lastRows = rows;

    if (!rows.length) {
      const tr = document.createElement("tr");
      const td = document.createElement("td");

      td.className = "empty-cell";

      if (currentPage === "teachers") {
        td.textContent =
          "No teacher records found. Register a teacher above.";
      } else if (currentPage === "students") {
        td.textContent =
          "No student records found. Register a student above.";
      } else {
        td.textContent = "No records found.";
      }

      tr.appendChild(td);
      body.appendChild(tr);

      $("record-count").textContent = "0 records";

      return;
    }

    const keys = [
      ...new Set(
        rows.flatMap(row => Object.keys(row || {}))
      )
    ];

    const headerRow = document.createElement("tr");

    keys.forEach(key => {
      const th = document.createElement("th");

      th.textContent = key.replaceAll("_", " ");

      headerRow.appendChild(th);
    });

    if (
      currentPage === "students" ||
      currentPage === "teachers"
    ) {
      const th = document.createElement("th");

      th.textContent = "Actions";

      headerRow.appendChild(th);
    }

    head.appendChild(headerRow);

    rows.forEach(row => {
      const tr = document.createElement("tr");

      keys.forEach(key => {
        const td = document.createElement("td");
        const value = row?.[key];

        td.textContent =
          value === null || value === undefined
            ? "—"
            : typeof value === "object"
              ? JSON.stringify(value)
              : String(value);

        tr.appendChild(td);
      });

      if (
        currentPage === "students" ||
        currentPage === "teachers"
      ) {
        const td = document.createElement("td");

        const editButton = document.createElement("button");

        editButton.type = "button";
        editButton.textContent = "Edit";
        editButton.className = "button secondary";

        editButton.dataset.recordAction = "edit";
        editButton.dataset.recordType = currentPage;
        editButton.dataset.id = row.id;

        const deleteButton = document.createElement("button");

        deleteButton.type = "button";
        deleteButton.textContent = "Delete";
        deleteButton.className = "button";
        deleteButton.style.marginLeft = "6px";

        deleteButton.dataset.recordAction = "delete";
        deleteButton.dataset.recordType = currentPage;
        deleteButton.dataset.id = row.id;

        td.append(editButton, deleteButton);

        tr.appendChild(td);
      }

      body.appendChild(tr);
    });

    $("record-count").textContent =
      `${rows.length} record${rows.length === 1 ? "" : "s"}`;
  }

  // ==========================================
  // LOAD MODULES
  // ==========================================

  async function loadModule(page) {
    currentPage = page;

    const [heading, description] = labels[page];

    $("page-title").textContent = heading;
    $("module-heading").textContent = heading;
    $("module-description").textContent = description;

    setMessage("Loading records...");

    try {
      const payload = await apiRequest(API[page]);
      const rows = extractRows(payload);

      renderTable(rows);

      setMessage(
        rows.length
          ? "Records loaded successfully."
          : "The connection is working, but there are no records yet.",
        "success"
      );

    } catch (error) {
      renderTable([]);

      setMessage(
        error.message || "Could not load this module.",
        "error"
      );
    }
  }

  // ==========================================
  // PAGE NAVIGATION
  // ==========================================

  function showPage(page) {
    document.querySelectorAll(".nav-link").forEach(button => {
      button.classList.toggle(
        "active",
        button.dataset.page === page
      );
    });

    if (page === "dashboard") {
      currentPage = "dashboard";

      $("page-title").textContent = "Dashboard";

      $("dashboard-page").hidden = false;
      $("dashboard-page").classList.add("active");

      $("module-page").hidden = true;

      loadDashboardStats();

    } else if (labels[page]) {

      $("dashboard-page").hidden = true;
      $("dashboard-page").classList.remove("active");

      $("module-page").hidden = false;
      $("module-page").classList.add("active");

      $("student-form-panel").hidden =
        page !== "students";

      $("teacher-form-panel").hidden =
        page !== "teachers";

      if (page === "students") {
        resetStudentForm();
      }

      if (page === "teachers") {
        resetTeacherForm();
      }

      loadModule(page);
    }

    $("sidebar").classList.remove("open");
  }

  // ==========================================
  // DASHBOARD STATISTICS
  // ==========================================

  async function loadDashboardStats() {
    const ids = {
      students: "stat-students",
      teachers: "stat-teachers",
      classes: "stat-classes",
      subjects: "stat-subjects"
    };

    for (const [module, id] of Object.entries(ids)) {
      try {
        const payload = await apiRequest(API[module]);

        $(id).textContent =
          extractRows(payload).length;

      } catch {
        $(id).textContent = "—";
      }
    }
  }

    const actionButton =
      event.target.closest("[data-record-action]");

    if (!actionButton) return;

    const id = actionButton.dataset.id;
    const type = actionButton.dataset.recordType;
    const action = actionButton.dataset.recordAction;

    if (type === "students" && action === "edit") {
      editStudent(id);
    }

    if (type === "students" && action === "delete") {
      deleteStudent(id);
    }

    if (type === "teachers" && action === "edit") {
      editTeacher(id);
    }

    if (type === "teachers" && action === "delete") {
      deleteTeacher(id);
    }
  });

  // ==========================================

    const button = event.target.closest("[data-page]");

    if (button) {
      showPage(button.dataset.page);

    });
