<script>
const $ = id => document.getElementById(id);
const state = {students:[],teachers:[],classes:[],subjects:[],attendance:[],examinations:[]};

async function api(url, options={}) {
  const response = await fetch(url, {
    ...options,
    headers: {"Content-Type":"application/json", ...(options.headers||{})}
  });
  let result;
  try { result = await response.json(); }
  catch { throw new Error("The server returned an invalid response."); }
  if (!response.ok || result.success === false) {
    throw new Error(result.message || `Request failed (${response.status})`);
  }
  return result;
}

function dataOf(result) {
  if (Array.isArray(result)) return result;
  return result && Array.isArray(result.data) ? result.data : [];
}
function notify(id, text, error=false) {
  const el=$(id);
  if (!el) return;
  el.textContent=text;
  el.className="notice "+(error?"error":"success");
}
function cell(row, value) {
  row.insertCell().textContent = value == null ? "" : String(value);
}
function addRow(tableId, values) {
  const row=$(tableId).insertRow();
  values.forEach(value=>cell(row,value));
}
function emptyRow(tableId, columns, text="No records found.") {
  const body=$(tableId);
  body.replaceChildren();
  const row=body.insertRow();
  const td=row.insertCell();
  td.colSpan=columns;
  td.textContent=text;
}
function grade(total) {
  if(total>=80)return "A";
  if(total>=70)return "B";
  if(total>=60)return "C";
  if(total>=50)return "D";
  if(total>=40)return "E";
  return "F";
}

function showPage(page) {
  document.querySelectorAll(".page").forEach(el=>el.hidden=true);
  $("page-"+page).hidden=false;
  document.querySelectorAll("#navigation button").forEach(btn=>btn.classList.toggle("active",btn.dataset.page===page));
  $("page-title").textContent=({dashboard:"Dashboard",students:"Students",teachers:"Teachers & Staff",classes:"Classes",subjects:"Subjects",attendance:"Attendance",examinations:"Examinations",fees:"Fees & Payments",reports:"Reports",settings:"Settings"})[page]||page;
  location.hash=page;
  loadPage(page);
}
$("navigation").addEventListener("click",event=>{
  const button=event.target.closest("button[data-page]");
  if(button)showPage(button.dataset.page);
});

async function loadStudents() {
  state.students=dataOf(await api("/api/students"));
  renderStudents();
  $("student-count").textContent=state.students.length;
  fillExamChoices();
}
function renderStudents() {
  const q=$("student-search").value.trim().toLowerCase();
  const list=state.students.filter(s=>[s.student_id,s.first_name,s.last_name,s.class_name,s.guardian_name].join(" ").toLowerCase().includes(q));
  $("student-table").replaceChildren();
  if(!list.length)return emptyRow("student-table",6);
  list.forEach(s=>addRow("student-table",[s.student_id,`${s.first_name||""} ${s.last_name||""}`.trim(),s.gender,s.class_name,s.guardian_name,s.guardian_phone]));
}
$("student-search").addEventListener("input",renderStudents);
$("student-form").addEventListener("submit",async event=>{
  event.preventDefault();
  try {
    const body=Object.fromEntries(new FormData(event.currentTarget));
    const result=await api("/api/students",{method:"POST",body:JSON.stringify(body)});
    notify("student-message",result.message||"Student saved successfully.");
    event.currentTarget.reset();
    await loadStudents();
  } catch(error) {notify("student-message",error.message,true);}
});

async function loadTeachers() {
  state.teachers=dataOf(await api("/api/teachers"));
  $("teacher-count").textContent=state.teachers.length;
  $("teacher-table").replaceChildren();
  if(!state.teachers.length)return emptyRow("teacher-table",6);
  state.teachers.forEach(t=>addRow("teacher-table",[t.staff_id,`${t.first_name||""} ${t.last_name||""}`.trim(),t.phone,t.email,t.subject,t.qualification]));
}
$("teacher-form").addEventListener("submit",async event=>{
  event.preventDefault();
  try {
    const body=Object.fromEntries(new FormData(event.currentTarget));

    event.currentTarget.reset();
    await loadTeachers();
  } catch(error) {notify("teacher-message",error.message,true);}
});

async function loadClasses() {
  const result=await api("/api/classes");
  state.classes=dataOf(result);
  $("class-table").replaceChildren();
  if(!state.classes.length)return emptyRow("class-table",2);
  state.classes.forEach(item=>addRow("class-table",[item.class_name||item.name||item, item.description||""]));
}
async function loadSubjects() {
  state.subjects=dataOf(await api("/api/subjects"));
  $("subject-count").textContent=state.subjects.length;
  $("subject-table").replaceChildren();
  if(!state.subjects.length)emptyRow("subject-table",3);
  else state.subjects.forEach(s=>addRow("subject-table",[s.subject_name,s.subject_code,s.description]));
  fillExamChoices();
}
$("subject-form").addEventListener("submit",async event=>{
  event.preventDefault();
  try {
    const body=Object.fromEntries(new FormData(event.currentTarget));
    const result=await api("/api/subjects",{method:"POST",body:JSON.stringify(body)});
    notify("subject-message",result.message||"Subject saved successfully.");
    event.currentTarget.reset();
    await loadSubjects();
  } catch(error) {notify("subject-message",error.message,true);}
});

async function loadAttendance() {
  const result=await api("/api/attendance");
  state.attendance=dataOf(result);
  $("attendance-table").replaceChildren();
  if(!state.attendance.length)return emptyRow("attendance-table",4);
  state.attendance.forEach(a=>addRow("attendance-table",[
    a.student_name||[a.first_name,a.last_name].filter(Boolean).join(" ")||a.student_id,
    a.class_name,a.attendance_date||a.date,a.status
  ]));
}

function fillExamChoices() {
  if(!$("exam-student"))return;
  const studentSelect=$("exam-student"),subjectSelect=$("exam-subject");
  const oldStudent=studentSelect.value,oldSubject=subjectSelect.value;
  studentSelect.replaceChildren(new Option("Select student",""));
  state.students.forEach(s=>studentSelect.add(new Option(`${s.student_id} — ${s.first_name} ${s.last_name}`,s.id)));
  subjectSelect.replaceChildren(new Option("Select subject",""));
  state.subjects.forEach(s=>subjectSelect.add(new Option(`${s.subject_name}${s.subject_code?" ("+s.subject_code+")":""}`,s.id)));
  if(state.students.some(s=>String(s.id)===oldStudent))studentSelect.value=oldStudent;
  if(state.subjects.some(s=>String(s.id)===oldSubject))subjectSelect.value=oldSubject;
  setExamClass();
}
function setExamClass() {
  const student=state.students.find(s=>String(s.id)===$("exam-student").value);
  $("exam-class").value=student?.class_name||"";
}
$("exam-student").addEventListener("change",setExamClass);
function updateExamPreview() {
  const a=$("class-score").value,b=$("exam-score").value;
  if(a===""||b===""){$("exam-total").value="0";$("exam-grade").value="—";return;}
  const c=Number(a),d=Number(b);
  if(!Number.isFinite(c)||!Number.isFinite(d)||c<0||c>30||d<0||d>70){$("exam-total").value="Invalid";$("exam-grade").value="—";return;}
  const total=c+d;
  $("exam-total").value=total.toFixed(2);
  $("exam-grade").value=grade(total);
}
$("class-score").addEventListener("input",updateExamPreview);
$("exam-score").addEventListener("input",updateExamPreview);

async function loadExaminations() {
  state.examinations=dataOf(await api("/api/examinations"));
  $("exam-count").textContent=state.examinations.length;
  renderExaminations();
}
function renderExaminations() {
  const q=$("exam-search").value.trim().toLowerCase();
  const rows=state.examinations.filter(e=>[
    e.student_number,e.first_name,e.last_name,e.subject_name,e.class_name,e.exam_name,e.academic_year,e.term,e.grade
  ].join(" ").toLowerCase().includes(q));
  $("exam-table").replaceChildren();
  if(!rows.length)return emptyRow("exam-table",11);
  rows.forEach(e=>addRow("exam-table",[
    e.student_number||"",`${e.first_name||""} ${e.last_name||""}`.trim(),
    e.class_name,e.subject_name,e.exam_name,e.academic_year,e.term,
    e.class_score,e.exam_score,e.total_score,e.grade
  ]));
}
$("exam-search").addEventListener("input",renderExaminations);
$("exam-form").addEventListener("submit",async event=>{
  event.preventDefault();
  const button=event.submitter;
  if(button)button.disabled=true;
  try {
    const body=Object.fromEntries(new FormData(event.currentTarget));
    body.student_id=Number(body.student_id);
    body.subject_id=Number(body.subject_id);
    body.class_score=Number(body.class_score);
    body.exam_score=Number(body.exam_score);
    const result=await api("/api/examinations",{method:"POST",body:JSON.stringify(body)});
    notify("exam-message",result.message||"Examination result saved successfully.");
    event.currentTarget.reset();
    $("exam-class").value="";
    updateExamPreview();
    await loadExaminations();
  } catch(error) {notify("exam-message",error.message,true);}
  finally {if(button)button.disabled=false;}
});
$("exam-form").addEventListener("reset",()=>setTimeout(()=>{ $("exam-class").value="";updateExamPreview();},0));

const loaders={
  dashboard:async()=>{
    try {
      await Promise.allSettled([loadStudents(),loadTeachers(),loadSubjects(),loadExaminations()]);
      notify("dashboard-message","Dashboard data refreshed.");
    } catch(error) {notify("dashboard-message",error.message,true);}
  },
  students:loadStudents,
  teachers:loadTeachers,
  classes:loadClasses,
  subjects:loadSubjects,
  attendance:loadAttendance,
  examinations:async()=>{await Promise.all([loadStudents(),loadSubjects(),loadExaminations()]);},
  fees:async()=>{},
  reports:async()=>{},
  settings:async()=>{}
};
async function loadPage(page) {
  try { await (loaders[page]||async()=>{})(); }
  catch(error) {
    const messageId=({students:"student-message",teachers:"teacher-message",classes:"class-message",subjects:"subject-message",attendance:"attendance-message",examinations:"exam-message"})[page];
    if(messageId)notify(messageId,error.message,true);
    console.error(`Could not load ${page}:`,error);
  }
}
$("refresh-button").addEventListener("click",()=>loadPage(location.hash.slice(1)||"dashboard"));

const initial=location.hash.slice(1);
showPage(["dashboard","students","teachers","classes","subjects","attendance","examinations","fees","reports","settings"].includes(initial)?initial:"dashboard");
